project(pseudopod)
cmake_minimum_required(VERSION 3.18)

set(CMAKE_VERBOSE_MAKEFILE ON)
set(FETCHCONTENT_UPDATES_DISCONNECTED ON CACHE BOOL "")
set(FETCHCONTENT_QUIET FALSE)
Include(FetchContent)

add_library(pseudopod INTERFACE)

########################
# CATCH2
########################

message(STATUS "Downloading Catch2 v3p3p2")
FetchContent_Declare(
	Catch2
	GIT_REPOSITORY https://github.com/catchorg/Catch2.git
	GIT_TAG        v3.3.2
  GIT_PROGRESS TRUE
	)
FetchContent_MakeAvailable(Catch2)
include_directories(${catch2_SOURCE_DIR}/src)

configure_file(pseudopod_config.h.in pseudopod_config.h)

install(DIRECTORY share/ DESTINATION share/pseudopod)

########################
# ZLIB
########################

find_package(ZLIB REQUIRED)
target_link_libraries(inq INTERFACE ZLIB::ZLIB)

########################
# LIBPAW INTERFACE
########################

message("\nConfiguring libPAW interface for pseudopod\n")
target_include_directories(pseudopod INTERFACE ${PROJECT_SOURCE_DIR}/external_libs/libpaw_interface/)
target_include_directories(pseudopod INTERFACE ${PROJECT_BINARY_DIR}/external_libs/libpaw_interface/)
message("\nDONE CONFIGURING LIBPAW INTERFACE\n")
set(libpaw_interface_include_dirs ${PROJECT_SOURCE_DIR}/external_libs/libpaw_interface)

target_link_libraries(pseudopod INTERFACE libpaw_interface)


############################
# UNIT TESTS
############################

file(GLOB INCLUDE_FILES RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} *hpp */*.hpp)

add_custom_command(OUTPUT pseudopod_unit_tests.cpp
													COMMAND echo ARGS //This file is automatically generated, do not edit > ${CMAKE_CURRENT_BINARY_DIR}/pseudopod_unit_tests.cpp)
													
foreach(TEST_FILE ${INCLUDE_FILES})
  get_filename_component(dir ${TEST_FILE} DIRECTORY)
  get_filename_component(name ${TEST_FILE} NAME_WE)
	string(TOUPPER INQ_${dir}_${name}_UNIT_TEST test_macro)
  add_custom_command(OUTPUT pseudopod_unit_tests.cpp
    COMMAND echo ARGS \\\#define ${test_macro} >> ${CMAKE_CURRENT_BINARY_DIR}/pseudopod_unit_tests.cpp
    COMMAND echo ARGS \\\#define INQ_TEST_FILE \\\"${TEST_FILE}\\\" >> ${CMAKE_CURRENT_BINARY_DIR}/pseudopod_unit_tests.cpp
    COMMAND echo ARGS \\\#define INQ_TEST_TAG \\\"[${name}]\\\" >> ${CMAKE_CURRENT_BINARY_DIR}/pseudopod_unit_tests.cpp
    COMMAND echo ARGS \\\#include \\\"${CMAKE_CURRENT_SOURCE_DIR}/${TEST_FILE}\\\" >> ${CMAKE_CURRENT_BINARY_DIR}/pseudopod_unit_tests.cpp
    COMMAND echo ARGS \\\#ifdef ${test_macro} >> ${CMAKE_CURRENT_BINARY_DIR}/pseudopod_unit_tests.cpp
    COMMAND echo ARGS \\\#error NO TEST INCLUDED, MACRO ${test_macro} STILL DEFINED >> ${CMAKE_CURRENT_BINARY_DIR}/pseudopod_unit_tests.cpp
    COMMAND echo ARGS \\\#endif >> ${CMAKE_CURRENT_BINARY_DIR}/pseudopod_unit_tests.cpp
    COMMAND echo ARGS \\\#undef INQ_TEST_FILE >> ${CMAKE_CURRENT_BINARY_DIR}/pseudopod_unit_tests.cpp
    COMMAND echo ARGS \\\#undef INQ_TEST_TAG >> ${CMAKE_CURRENT_BINARY_DIR}/pseudopod_unit_tests.cpp
		DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/${TEST_FILE}
		APPEND)

	add_test(NAME pseudopod::unit_tests::${dir}::${name} COMMAND ./pseudopod_unit_tests [${name}])
	set_property(TEST pseudopod::unit_tests::${dir}::${name} PROPERTY ENVIRONMENT_MODIFICATION PSEUDOPOD_SHARE_PATH=set:${PROJECT_SOURCE_DIR}/share)
endforeach()

include_directories(${PROJECT_SOURCE_DIR}/external_libs)
include_directories(${PROJECT_BINARY_DIR})

if(ENABLE_GPU)
	set_source_files_properties(pseudopod_unit_tests.cpp PROPERTIES LANGUAGE ${GPU_LANGUAGE})
endif()

add_executable(pseudopod_unit_tests)
target_sources(pseudopod_unit_tests PRIVATE pseudopod_unit_tests.cpp)
target_link_libraries(pseudopod_unit_tests PRIVATE Catch2::Catch2WithMain inq)
